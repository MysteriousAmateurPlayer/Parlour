{{- /* 正文里图片的渲染钩子 把 ![](/images/x.png) 自动补上 /仓库名/ 前缀 —— 写作台上传的图片 插入的就是 /images/xxx.png 这种写法，靠这个钩子才能正确显示。 相对路径的图片也按当前页面地址解析。 */ -}} {{- $dest := .Destination -}} {{- $src := $dest -}} {{- if and (not (hasPrefix $dest "http")) (not (hasPrefix $dest "//")) (not (hasPrefix $dest "data:")) -}} {{- if hasPrefix $dest "/" -}} {{- $src = partial "link.html" $dest -}} {{- else -}} {{- $src = path.Join .Page.RelPermalink $dest -}} {{- end -}} {{- end -}}
